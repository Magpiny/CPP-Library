111template <std::floating_point T = double, PolygonRange R>
112 requires std::same_as<std::ranges::range_value_t<R>, Point2D<T>>
113[[nodiscard]] constexpr std::optional<Area<T>>
114calculate_area_shoelace(const R &vertices) {
115 const std::size_t n = std::ranges::size(vertices);
116
117 if (n < 3) {
118 return std::nullopt;
119 }
120
121 T sum{};
122
123 for (std::size_t i = 0; i < n; ++i) {
124 const auto &current = vertices[i];
125 const auto &next = vertices[(i + 1) % n];
126
127 sum += current.x.value * next.y.value;
128 sum -= next.x.value * current.y.value;
129 }
130
131 return Area<T>{std::abs(sum) / T{2}};
132}

140template <std::floating_point T = double, PolygonRange R>

142[[nodiscard]] constexpr std::optional<Perimeter<T>>

144 const std::size_t n = std::ranges::size(vertices);

146 if (n < 3) {
147 return std::nullopt;
148 }
149
150 T total{};
151
152 for (std::size_t i = 0; i < n; ++i) {
153 const auto &current = vertices[i];
154 const auto &next = vertices[(i + 1) % n];

172template <std::floating_point T = double>
173[[nodiscard]] constexpr std::optional<Area<T>>
174calculate_regular_polygon_area(std::size_t num_sides, T side_length) noexcept {
175
176 if (num_sides < 3 || side_length <= T{}) {
177 return std::nullopt;
178 }
179
180 const T n = static_cast<T>(num_sides);
181 const T angle = std::numbers::pi_v<T> / n;
182 const T area_val = (n * side_length * side_length) / (T{4} * std::tan(angle));
183
184 return Area<T>{area_val};
185}
